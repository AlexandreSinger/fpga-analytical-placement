# Read in the flat placement and stop after global placement (--global_place),
# which skips the global placer, so only the estimates of the read-in flat
# placement are made. Writes the estimated timing graph, wire usage, and
# legality echo files.
# The router options are kept the same as the routed task, since the place
# delay model used by the estimate is built from the router lookahead.
script_params=-starting_stage vpr --route_chan_width 400 --max_router_iterations 400 --initial_pres_fac 1.0 --router_profiler_astar_fac 1.5 -timeout 86400 --seed 1 --global_place --echo_file timing_graph.ap_post_routing_estimate.echo wire_usage.ap_post_routing_estimate.echo legality.ap_estimate.echo
