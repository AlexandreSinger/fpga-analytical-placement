##############################################
# Configuration file for running experiments
##############################################
#
# Global placement: runs only the global placement stage of the AP flow
# (--global_place), prints the legality estimate of the global placement
# ("Legality Estimate Summary"), writes its per-tile penalty to
# legality.ap_estimate.echo, and writes the global placement to gp.fplace.
#
# The circuits, devices, and router options are the same as the artix_7_ap
# test suite
# (fpga-analytical-placement/testing/test_suites/artix_7_ap): the VTR
# circuits on Artix-7-like devices of the 7-series architecture, routed at
# the physical channel width of the chip (190).
#
# See ../../README.md for how to run this experiment.

# Path to directory of circuits to use
circuits_dir=/shared/singera8/7series_blifs

# Path to directory of architectures to use
archs_dir=arch/xilinx

# Add architectures to list to sweep
arch_list_add=7series_BRAM_DSP_carry.xml

# Add circuits to list to sweep
circuit_list_add=arm_core.blif
circuit_list_add=bgm.blif
circuit_list_add=stereovision0.blif
circuit_list_add=stereovision1.blif
circuit_list_add=stereovision2.blif
circuit_list_add=LU8PEEng.blif
circuit_list_add=LU32PEEng.blif
circuit_list_add=mcml.blif

# Constrain the circuits to their devices
circuit_constraint_list_add=(arm_core.blif, device=XC7A200T-like)
circuit_constraint_list_add=(bgm.blif, device=XC7A75T-like)
circuit_constraint_list_add=(stereovision0.blif, device=XC7A200T-like)
circuit_constraint_list_add=(stereovision1.blif, device=XC7A75T-like)
circuit_constraint_list_add=(stereovision2.blif, device=XC7A200T-like)
circuit_constraint_list_add=(LU8PEEng.blif, device=XC7A50T-like)
circuit_constraint_list_add=(LU32PEEng.blif, device=XC7A200T-like)
circuit_constraint_list_add=(mcml.blif, device=XC7A200T-like)

# Parse info and how to parse
parse_file=vpr_ap_global_place.txt

# How to parse QoR info
qor_parse_file=qor_ap_fixed_chan_width.txt

# Pass requirements
pass_requirements_file=pass_requirements_ap_global_place.txt

# Run only global placement, and write the legality echo file and the global
# placement. The router options are kept the same as the full AP flow, since
# the place delay model used by the timing-driven global placer is built from
# the router lookahead.
script_params=-starting_stage vpr --route_chan_width 190 --seed 1 -track_memory_usage --global_place --write_post_gp_flat_place gp.fplace --echo_file legality.ap_estimate.echo
